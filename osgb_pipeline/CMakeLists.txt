cmake_minimum_required(VERSION 3.16)
project(osgb_extract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# 本机没有安装 libopenscenegraph-dev 的系统头文件时，使用项目 deps/ 中
# 解包的 Ubuntu 官方开发包头文件；运行库仍链接系统已有的 OSG 3.6.5。
set(OSG_LOCAL_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/deps/root/usr/include")
if(EXISTS "${OSG_LOCAL_INCLUDE}/osg/Node")
    if(NOT OSG_INCLUDE_DIR)
        set(OSG_INCLUDE_DIR "${OSG_LOCAL_INCLUDE}" CACHE PATH "OpenSceneGraph include directory")
    endif()
    if(NOT OSGDB_INCLUDE_DIR)
        set(OSGDB_INCLUDE_DIR "${OSG_LOCAL_INCLUDE}" CACHE PATH "osgDB include directory")
    endif()
    if(NOT OPENTHREADS_INCLUDE_DIR)
        set(OPENTHREADS_INCLUDE_DIR "${OSG_LOCAL_INCLUDE}" CACHE PATH "OpenThreads include directory")
    endif()
endif()

if(NOT OSG_LIBRARY_RELEASE)
    find_library(OSG_LIBRARY_RELEASE NAMES osg PATHS /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
endif()
if(NOT OSGDB_LIBRARY_RELEASE)
    find_library(OSGDB_LIBRARY_RELEASE NAMES osgDB PATHS /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
endif()
if(NOT OPENTHREADS_LIBRARY_RELEASE)
    find_library(OPENTHREADS_LIBRARY_RELEASE NAMES OpenThreads PATHS /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
endif()

find_package(OpenSceneGraph REQUIRED COMPONENTS osg osgDB OpenThreads)
add_executable(extract_osgb src/extract_osgb.cpp)
target_include_directories(extract_osgb PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(extract_osgb PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
add_executable(extract_las src/extract_las.cpp)
target_include_directories(extract_las PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(extract_las PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
add_executable(make_fixture src/make_fixture.cpp)
target_include_directories(make_fixture PRIVATE ${OPENSCENEGRAPH_INCLUDE_DIRS})
target_link_libraries(make_fixture PRIVATE ${OPENSCENEGRAPH_LIBRARIES})
